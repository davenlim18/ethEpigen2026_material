df_agree <- data.frame(
  DMR_index = queryHits(hits),
  DNAme_change = dmr_chr1$meandiff[queryHits(hits)],
  ATAC_change = mcols(da_ranges)[[atac_change_col]][subjectHits(hits)]
)

df_agree <- df_agree[complete.cases(df_agree), ]

df_agree$relationship <- ifelse(
  sign(df_agree$DNAme_change) == -sign(df_agree$ATAC_change),
  "opposite direction",
  "same direction"
)
ggplot(df_agree, aes(DNAme_change, ATAC_change)) +
  geom_point(aes(colour = relationship), alpha = 0.7) +
  geom_hline(yintercept = 0, linetype = "dashed") +
  geom_vline(xintercept = 0, linetype = "dashed") +

This overlaps DMRs with differential accessibility regions and compares methylation change with ATAC change at the same loci. Opposite-direction changes suggest expected regulatory agreement, where increased methylation corresponds to reduced accessibility, while same-direction changes suggest disagreement or more complex regulation.
